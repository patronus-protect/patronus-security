## Reproduktion

```bash
D="$ARK_L1_DEV_DATA"
for f in $(cd "$D" && ls *.jsonl */*.jsonl); do
  for m in strict balanced; do
    mkdir -p "/tmp/tri/ark-$m/$(dirname "$f")"
    .venv/bin/python scripts/l1_presidio/run_ark.py --pii-mode "$m" --input "$D/$f" --output "/tmp/tri/ark-$m/$f"
  done
  mkdir -p "/tmp/tri/presidio-tuned/$(dirname "$f")"
  ~/ark-l1/presidio-venv/bin/python scripts/l1_presidio/run_presidio.py --profile tuned \
    --input "$D/$f" --output "/tmp/tri/presidio-tuned/$f"
done
for cmd in m1 m2; do
  python3 scripts/l1_presidio/scoreboard.py "$cmd" --gold "$D" --ark strict=/tmp/tri/ark-strict \
    --ark balanced=/tmp/tri/ark-balanced --presidio /tmp/tri/presidio-tuned
done
python3 scripts/l1_presidio/scoreboard.py report --gold "$D" --ark strict=/tmp/tri/ark-strict \
  --ark balanced=/tmp/tri/ark-balanced --engine presidio-tuned=/tmp/tri/presidio-tuned
```
